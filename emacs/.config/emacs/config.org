(defun my/gptel-refactor-region ()
  "Bypasses the gptel menu. Sends the selected region directly to DeepSeek 
with the refactoring persona."
  (interactive)
  (unless (use-region-p)
    (user-error "Select a block of code to refactor first."))
  
  ;; Dynamically bind the specific backend, model, and directive for this execution only
  (let ((gptel-backend gptel-openrouter)
        (gptel-model "deepseek/deepseek-r2")
        (gptel-system-message (alist-get 'refactor gptel-directives)))
    
    ;; Execute the send command on the active region
    (gptel-send)))

;; Bind it to C-c c r (Code Refactor)
(define-key global-map (kbd "C-c c r") #'my/gptel-refactor-region)
#+end_src

#+begin_src elisp
(use-package elfeed
  :ensure t
  :bind (("C-c l e" . elfeed)
         ("C-c l u" . elfeed-update)
         :map elfeed-show-mode-map
         ("b" . my/elfeed-abrir-no-eww))
  :preface
  (defun my/elfeed-abrir-no-eww ()
    "Abre o artigo atual no EWW (browser interno) em vez do browser externo."
    (interactive)
    (let ((link (elfeed-entry-link elfeed-show-entry)))
      (when link
        (message "A descarregar o artigo completo...")
        (eww link))))
  :config
  (setq elfeed-use-curl t)
  ;; Atualiza o Elfeed a cada 30 minutos em background
  (run-at-time nil (* 30 60) #'elfeed-update))

(use-package elfeed-protocol
  :ensure t
  :after elfeed
  :config
  (setq elfeed-protocol-fever-update-unread-only t)
  
  (setq elfeed-feeds
        '(("fever+https://gabi@vm-magasirala-rss.duckdns.org"
           :api-url "https://vm-magasirala-rss.duckdns.org/api/fever.php"
           :use-authinfo t
           :autotags ("freshrss"))))
  (elfeed-protocol-enable))
#+end_src
