  (defun my/gptel-fix-region-or-buffer ()
    "Abre uma sessão dedicada e pede a correção da região (ou do buffer)."
    (interactive)
    (let* ((beg (if (use-region-p) (region-beginning) (point-min)))
           (end (if (use-region-p) (region-end) (point-max)))
           (lang (string-remove-suffix "-ts" (string-remove-suffix "-mode" (symbol-name major-mode))))
           (code (buffer-substring-no-properties beg end))
           (name "*gptel-fix*"))
      (gptel name)
      (with-current-buffer name
        (goto-char (point-max))
        ;; Concatenado para impedir que o Babel extraia "#+end_src" no meio da string!
        (insert "Fix this code:\n\n#+begin_src " lang "\n" code (concat "\n#+end" "_src\n"))
        (gptel-send))

  ;; --- SISTEMA ROBUSTO DE SNAPSHOT GIT ---
  (defun my/git (&rest args)
    "Executa git sincronamente devolvendo a string."
    (with-temp-buffer
      (let ((rc (apply #'process-file "git" nil t nil args)))
        (unless (zerop rc) (error "git falhou: %s" (string-trim (buffer-string))))
        (string-trim (buffer-string)))))

  (defvar my/gptel-last-checkpoint nil)

  (defun my/gptel-git-checkpoint-create ()
    "Cria um snapshot do estado atual sem alterar a working tree ou o git stash"
    (interactive)
    (let* ((default-directory (my/project-root-strict))
           (idx (expand-file-name (make-temp-name "gptel-idx-") temporary-file-directory)))
      (unwind-protect
          (let ((process-environment (cons (concat "GIT_INDEX_FILE=" idx) process-environment)))
            (my/git "read-tree" "HEAD")
            (my/git "add" "-A")
            (let* ((tree (my/git "write-tree"))
                   (commit (my/git "commit-tree" tree "-p" "HEAD" "-m" "gptel checkpoint"))
                   (ref (format-time-string "refs/gptel/%Y%m%d-%H%M%S")))
              (my/git "update-ref" ref commit)
              (setq my/gptel-last-checkpoint commit)
              (message "🔖 Snapshot seguro gerado em: %s @ %s" ref (substring commit 0 8))))
        (when (file-exists-p idx) (delete-file idx)))))

  (defun my/gptel-git-checkpoint-rollback ()
    "Restaura o repositório para o exato momento antes da execução do agente"
    (interactive)
    (unless my/gptel-last-checkpoint (user-error "Nenhum snapshot em memória para restaurar!"))
    (let* ((default-directory (my/project-root-strict))
           (cp my/gptel-last-checkpoint))
      (when (yes-or-no-p (format "⚠️ Restaurar o repositório todo para %s? " (substring cp 0 8)))
        (my/git "restore" (concat "--source=" cp) "--worktree" "--" ":/")
        ;; Remove ficheiros extra que o LLM tenha criado
        (let ((snap (split-string (my/git "ls-tree" "-r" "--name-only" cp) "\n" t)))
          (dolist (f (split-string (my/git "ls-files" "--others" "--exclude-standard") "\n" t))
            (unless (member f snap) (delete-file f))))
        (message "⏪ Repositório restaurado limpo para %s" (substring cp 0 8)))))

  (defun my/gptel-review-diff ()
    (interactive)
    (let ((default-directory (my/project-root-strict)))
      (if my/gptel-last-checkpoint
       (progn
          (magit-diff-range my/gptel-last-checkpoint)
          (message "Ficheiros novos/não rastreados: %s"
                   (mapconcat #'identity
                              (split-string (my/git "-c" "core.quotePath=false" "ls-files" "-z" "--others" "--exclude-standard") "\0" t)
                              ", ")))
      (user-error "Nenhum checkpoint ativo do agente para comparar!"))))

  ;; --- PLAN & ACT WORKFLOW ---
  (defun my/gptel-plan (prompt)
    (interactive "sPrompt para o Plano (ex: Adicionar parse JSON): ")
    (let* ((root (my/project-root-strict))
           (plan-file (expand-file-name "PLAN.org" root))
           (plan-buffer (find-file-noselect plan-file)))
      (with-current-buffer plan-buffer
        (goto-char (point-max))
        (insert (format "\n* Plan Request [%s]: %s\n\n" 
                        (format-time-string "%Y-%m-%d %H:%M:%S") prompt))
        (save-buffer))
      (pop-to-buffer plan-buffer)
      (gptel-with-preset 'plan-mode
        (gptel-request (format "Analisa o repositório e elabora o PLAN.org para:\n%s" prompt)
          :buffer plan-buffer
          :position (copy-marker (point-max) t)
          :stream t))
      (message "🧠 Agente em PLAN MODE (READ-ONLY). A gerar...")))

  (defun my/gptel-execute-plan ()
    (interactive)
    (let* ((root (my/project-root-strict))
           (plan-file (expand-file-name "PLAN.org" root)))
      (unless (file-exists-p plan-file) (user-error "Não existe nenhum PLAN.org!"))
      (when (y-or-n-p "🚀 Criar Snapshot Automático e EXECUTAR o PLAN.org? ")
        (my/gptel-git-checkpoint-create)
        (let ((plan-content (with-temp-buffer (insert-file-contents plan-file) (buffer-string)))
              (agent-buf (get-buffer-create "*gptel-executor*")))
          (pop-to-buffer agent-buf)
          (unless (derived-mode-p 'text-mode) (org-mode))
          (gptel-mode 1)
          (goto-char (point-max))
          (insert "\n=== A INICIAR EXECUÇÃO DO PLANO ===\n\n")
          (gptel-with-preset 'act-mode
            (gptel-request (concat "Executa rigorosamente este PLAN.org:\n\n" plan-content)
              :buffer agent-buf
              :position (copy-marker (point-max) t)
              :stream t))
          (message "⚡ ACT MODE Ativado. Aguardando ferramentas..."))))))
